cmake_minimum_required(VERSION 3.22)
project(FlyAuditory VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(FLY_BUILD_PLUGIN "Build JUCE VST3 and Standalone" ON)
option(FLY_BUILD_TESTS "Build core and processor integration tests" ON)
set(JUCE_PATH "" CACHE PATH "Path to an existing JUCE 8 checkout")
add_library(FlyCore STATIC Source/DataLoader.cpp Source/AuditoryPathwaySimulator.cpp)
set_target_properties(FlyCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(FlyCore PUBLIC Source)
if(MSVC)
  target_compile_options(FlyCore PRIVATE /W4)
else()
  target_compile_options(FlyCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FLY_BUILD_PLUGIN)
  if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
    message(FATAL_ERROR "Set -DJUCE_PATH=/path/to/JUCE (JUCE 8). No network fetching occurs during builds.")
  endif()
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
  juce_add_binary_data(FlyData SOURCES Data/auditory.flygraph)
  juce_add_plugin(Fly
    COMPANY_NAME "Lazirko Records"
    BUNDLE_ID "com.lazirkorecords.flyauditory"
    PLUGIN_MANUFACTURER_CODE Lazi
    PLUGIN_CODE Flya
    FORMATS VST3 Standalone
    PRODUCT_NAME "Drosophila melanogaster"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Fx Filter Modulation)
  target_sources(Fly PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/Visualizer.cpp)
  target_include_directories(Fly PRIVATE Source)
  target_compile_definitions(Fly PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(Fly PRIVATE FlyCore FlyData juce::juce_audio_utils juce::juce_opengl
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endif()
if(FLY_BUILD_TESTS)
  enable_testing()
  add_executable(FlyBenchmark Tests/Benchmark.cpp)
  target_link_libraries(FlyBenchmark PRIVATE FlyCore)
  add_executable(FlyCoreTests Tests/CoreTests.cpp)
  target_link_libraries(FlyCoreTests PRIVATE FlyCore)
  add_test(NAME auditory_core COMMAND FlyCoreTests "${CMAKE_CURRENT_SOURCE_DIR}/Data/auditory.flygraph")
  if(FLY_BUILD_PLUGIN)
    juce_add_console_app(FlyPluginTests PRODUCT_NAME "Fly Processor Tests")
    target_sources(FlyPluginTests PRIVATE Tests/PluginTests.cpp)
    target_include_directories(FlyPluginTests PRIVATE Source)
    target_link_libraries(FlyPluginTests PRIVATE Fly FlyCore FlyData juce::juce_audio_utils juce::juce_opengl)
    add_test(NAME auditory_processor COMMAND FlyPluginTests)
    juce_add_console_app(FlyVst3HostTests PRODUCT_NAME "Fly VST3 Host Tests")
    target_sources(FlyVst3HostTests PRIVATE Tests/Vst3HostTests.cpp)
    target_compile_definitions(FlyVst3HostTests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(FlyVst3HostTests PRIVATE juce::juce_audio_utils juce::juce_opengl)
    add_dependencies(FlyVst3HostTests Fly_VST3)
    if(APPLE)
      set(FLY_VST3_TEST_PATH "$<TARGET_FILE_DIR:Fly_VST3>/../..")
    elseif(WIN32)
      set(FLY_VST3_TEST_PATH "$<TARGET_FILE_DIR:Fly_VST3>/../..")
    else()
      set(FLY_VST3_TEST_PATH "$<TARGET_FILE:Fly_VST3>")
    endif()
    add_test(NAME auditory_vst3_bundle COMMAND FlyVst3HostTests "${FLY_VST3_TEST_PATH}")
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME auditory_data COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_data.py")
  endif()
endif()
